> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Laravel CRM for managing WhatsApp leads interested in a land lot at ALTAI Residencial Cancun, with a REST API for N8N integration, pipeline kanban UI, and 3 users.

**Architecture:** Laravel 12 + Livewire 3 + Tailwind CSS. MySQL database with 4 tables (contactos, actividades, users, api_tokens). REST API authenticated via X-API-Key for N8N to create/update leads. Livewire-based UI with dashboard, pipeline kanban, and contact detail views.

**Tech Stack:** PHP 8.2+, Laravel 12, Livewire 3, Tailwind CSS, MySQL, PHPUnit

Create `app/Models/Contacto.php`:
```php
<?php

namespace App\Models;

use App\Enums\Etapa;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsTo;
use Illuminate\Database\Eloquent\Relations\HasMany;

class Contacto extends Model
{
    protected $table = 'contactos';

    protected $fillable = [
        'nombre', 'email', 'telefono', 'etapa', 'asignado_a',
        'notas', 'fuente', 'motivo_no_interes',
        'fecha_contacto', 'fecha_escalado', 'ultima_actividad_at',
    ];

    protected function casts(): array
    {
        return [
            'etapa' => Etapa::class,
            'fecha_contacto' => 'datetime',
            'fecha_escalado' => 'datetime',
            'ultima_actividad_at' => 'datetime',
        ];
    }

    /**
     * Normalize phone to E.164 format before saving.
     */
    public function setTelefonoAttribute(string $value): void
    {
        $digits = preg_replace('/\D/', '', $value);

        // If starts with 52 and has 12+ digits, add +
        if (str_starts_with($digits, '52') && strlen($digits) >= 12) {
            $this->attributes['telefono'] = '+' . $digits;
        }
        // If 10 digits (Mexican local), prepend +521
        elseif (strlen($digits) === 10) {
            $this->attributes['telefono'] = '+521' . $digits;
        }
        // Otherwise store as-is with +
        else {
            $this->attributes['telefono'] = '+' . ltrim($digits, '+');
        }
    }

    public function asignadoA(): BelongsTo
    {
        return $this->belongsTo(User::class, 'asignado_a');
    }

    public function actividades(): HasMany
    {
        return $this->hasMany(Actividad::class, 'contacto_id');
    }

    public function scopeEtapa($query, Etapa $etapa)
    {
        return $query->where('etapa', $etapa);
    }

    public function scopeInactivos($query, int $horas)
    {
        return $query->where('etapa', Etapa::PROSPECTO)
            ->where('ultima_actividad_at', '<', now()->subHours($horas));
    }

    public function scopeBuscar($query, string $term)
    {
        return $query->where(function ($q) use ($term) {
            $q->where('nombre', 'like', "%{$term}%")
              ->orWhere('email', 'like', "%{$term}%")
              ->orWhere('telefono', 'like', "%{$term}%");
        });
    }

    /**
     * Normalize phone for search comparison.
     */
    public static function normalizarTelefono(string $phone): string
    {
        $digits = preg_replace('/\D/', '', $phone);
        if (str_starts_with($digits, '52') && strlen($digits) >= 12) {
            return '+' . $digits;
        }
        if (strlen($digits) === 10) {
            return '+521' . $digits;
        }
        return '+' . ltrim($digits, '+');
    }
}
```

- [ ] **Step 5: Create ContactoService (centralized stage-change logic)**

Create `app/Services/ContactoService.php`:
```php
<?php

namespace App\Services;

use App\Enums\Etapa;
use App\Enums\TipoActividad;
use App\Models\Contacto;

class ContactoService
{
    /**
     * Change a contact's stage with proper side effects and activity logging.
     * Used by API controller, KanbanBoard, and ContactoDetail to avoid duplication.
     */
    public function cambiarEtapa(Contacto $contacto, Etapa $etapaNueva, ?string $notas = null, bool $automatico = true): Contacto
    {
        $etapaAnterior = $contacto->etapa;

        $contacto->etapa = $etapaNueva;

        if ($etapaNueva === Etapa::INTERESADO) {
            $contacto->fecha_escalado = now();
        }

        if ($etapaNueva === Etapa::NO_INTERESADO && $notas) {
            $contacto->motivo_no_interes = $notas;
        }

        $contacto->save();

        $contacto->actividades()->create([
            'tipo' => TipoActividad::CAMBIO_ETAPA,
            'descripcion' => "Cambio de {$etapaAnterior->label()} a {$etapaNueva->label()}",
            'metadata' => [
                'etapa_anterior' => $etapaAnterior->value,
                'etapa_nueva' => $etapaNueva->value,
                'notas' => $notas,
            ],
            'automatico' => $automatico,
        ]);

        return $contacto->fresh();
    }
}
```

```php
<?php

namespace App\Http\Middleware;

use App\Models\ApiToken;
use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

class ApiTokenAuth
{
    public function handle(Request $request, Closure $next): Response
    {
        $token = $request->header('X-API-Key');

        if (!$token) {
            return response()->json(['error' => 'API key requerida.'], 401);
        }

        $apiToken = ApiToken::where('token', $token)
            ->where('active', true)
            ->first();

        if (!$apiToken) {
            return response()->json(['error' => 'API key inválida.'], 401);
        }

        if ($apiToken->expires_at && $apiToken->expires_at->isPast()) {
            return response()->json(['error' => 'API key expirada.'], 401);
        }

        $apiToken->update(['last_used_at' => now()]);
        $request->attributes->set('api_token', $apiToken);

        return $next($request);
    }
}
```

namespace App\Http\Controllers\Api\V1;

use App\Enums\Etapa;

use App\Http\Controllers\Controller;
use App\Models\Contacto;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Validation\Rules\Enum;

class ContactoController extends Controller
{
    public function index(Request $request): JsonResponse
    {
        $query = Contacto::with('asignadoA');

        if ($request->has('etapa')) {
            $query->where('etapa', $request->etapa);
        }
        if ($request->has('asignado_a')) {
            $query->where('asignado_a', $request->asignado_a);
        }

        $contactos = $query->orderByDesc('created_at')
            ->paginate($request->integer('per_page', 25));

        return response()->json($contactos);
    }

    public function store(Request $request): JsonResponse
    {
        $validated = $request->validate([
            'nombre' => 'required|string|max:255',
            'email' => 'nullable|email|max:255',
            'telefono' => 'required|string|max:20',
            'fuente' => 'nullable|string|max:50',
            'notas' => 'nullable|string',
        ]);

        // Normalize phone and check for duplicates
        $telefonoNormalizado = Contacto::normalizarTelefono($validated['telefono']);
        $existente = Contacto::where('telefono', $telefonoNormalizado)->first();

        if ($existente) {
            return response()->json([
                'error' => 'Ya existe un contacto con este teléfono.',
                'contacto_id' => $existente->id,
            ], 409);
        }

        $contacto = Contacto::create($validated);

        return response()->json($contacto, 201);
    }

    public function show(Contacto $contacto): JsonResponse
    {
        return response()->json(
            $contacto->load(['asignadoA', 'actividades' => fn($q) => $q->orderByDesc('created_at')])
        );
    }

    public function update(Request $request, Contacto $contacto): JsonResponse
    {
        $validated = $request->validate([
            'nombre' => 'sometimes|string|max:255',
            'email' => 'sometimes|nullable|email|max:255',
            'notas' => 'sometimes|nullable|string',
            'asignado_a' => 'sometimes|nullable|exists:users,id',
        ]);

        $contacto->update($validated);

        return response()->json($contacto->fresh());
    }

    public function updateEtapa(Request $request, Contacto $contacto): JsonResponse
    {
        $validated = $request->validate([
            'etapa' => ['required', new Enum(Etapa::class)],
            'notas' => 'nullable|string',
        ]);

        $etapaNueva = Etapa::from($validated['etapa']);
        $contacto = app(\App\Services\ContactoService::class)
            ->cambiarEtapa($contacto, $etapaNueva, $validated['notas'] ?? null, true);

        return response()->json($contacto);
    }

    public function buscar(Request $request): JsonResponse
    {
        $request->validate(['telefono' => 'required|string']);

        $telefonoNormalizado = Contacto::normalizarTelefono($request->telefono);
        $contacto = Contacto::where('telefono', $telefonoNormalizado)->first();

        if (!$contacto) {
            return response()->json(['error' => 'No encontrado.'], 404);
        }

        return response()->json($contacto);
    }

    public function inactivos(Request $request): JsonResponse
    {
        $horas = $request->integer('horas', 48);

        $contactos = Contacto::inactivos($horas)
            ->select('id', 'nombre', 'telefono', 'ultima_actividad_at')
            ->get();

        return response()->json(['contactos' => $contactos]);
    }
}

class ContactoDetail extends Component
{
    public Contacto $contacto;
    public string $nuevaNota = '';

    public function mount(Contacto $contacto): void
    {
        $this->contacto = $contacto;
    }

    public function cambiarEtapa(string $etapa): void
    {
        $etapaNueva = Etapa::from($etapa);
        $this->contacto = app(\App\Services\ContactoService::class)
            ->cambiarEtapa($this->contacto, $etapaNueva, null, false);
    }

    public function agregarNota(): void
    {
        if (empty(trim($this->nuevaNota))) return;

        $this->contacto->actividades()->create([
            'tipo' => TipoActividad::NOTA,
            'descripcion' => $this->nuevaNota,
            'automatico' => false,
        ]);

        $this->contacto->update(['ultima_actividad_at' => now()]);
        $this->nuevaNota = '';
        $this->contacto->refresh();
    }

    public function render()
    {
        $actividades = $this->contacto->actividades()
            ->orderByDesc('created_at')
            ->get();

        return view('livewire.contactos.contacto-detail', compact('actividades'));
    }
}

Create `app/Livewire/Settings/ApiTokens.php`:
```php
<?php

namespace App\Livewire\Settings;

use App\Models\ApiToken;
use Livewire\Component;

class ApiTokens extends Component
{
    public string $tokenName = '';
    public ?string $newToken = null;

    public function createToken(): void
    {
        $this->validate(['tokenName' => 'required|string|max:255']);

        $token = ApiToken::create([
            'name' => $this->tokenName,
            'token' => ApiToken::generateToken(),
            'active' => true,
        ]);

        $this->newToken = $token->token;
        $this->tokenName = '';
    }

    public function toggleToken(int $id): void
    {
        $token = ApiToken::findOrFail($id);
        $token->update(['active' => !$token->active]);
    }

    public function deleteToken(int $id): void
    {
        ApiToken::findOrFail($id)->delete();
    }

    public function render()
    {
        return view('livewire.settings.api-tokens', [
            'tokens' => ApiToken::orderByDesc('created_at')->get(),
        ]);
    }
}
```
